Sort and paginate in the browser — no remote data API or row reorder in this phase.

<ComponentTabs>
  <Fragment slot="examples">
    <PondDemo label="Sort + pages">
      <pond-dynamic-table
        caption="People"
        page-size="3"
      ></pond-dynamic-table>
      <script>
        const table = document.currentScript.previousElementSibling
        table.columns = [
          { key: 'name', label: 'Name' },
          { key: 'role', label: 'Role' }
        ]
        table.rows = [
          { name: 'Ada', role: 'Lead' },
          { name: 'Bea', role: 'QA' },
          { name: 'Cal', role: 'Dev' },
          { name: 'Dee', role: 'PM' }
        ]
      </script>
    </PondDemo>
  </Fragment>

  <Fragment slot="code">

```html
<pond-dynamic-table
  caption="People"
  page-size="5"
></pond-dynamic-table>
<script type="module">
  const el = document.querySelector('pond-dynamic-table')
  el.columns = [
    { key: 'name', label: 'Name' },
    { key: 'role', label: 'Role' }
  ]
  el.rows = [
    { name: 'Ada', role: 'Lead' },
    { name: 'Bea', role: 'QA' }
  ]
</script>
```
