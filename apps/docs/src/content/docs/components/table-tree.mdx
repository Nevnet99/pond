Nested rows expand in place. No remote children API or drag-and-drop yet.

<ComponentTabs>
  <Fragment slot="examples">
    <PondDemo label="Expanded">
      <pond-table-tree caption="Project"></pond-table-tree>
      <script>
        const tree = document.currentScript.previousElementSibling
        tree.columns = [
          { key: 'name', label: 'Name' },
          { key: 'type', label: 'Type' }
        ]
        tree.nodes = [
          {
            id: 'src',
            cells: { name: 'src', type: 'Folder' },
            children: [{ id: 'a', cells: { name: 'app.ts', type: 'File' } }]
          },
          { id: 'r', cells: { name: 'README.md', type: 'File' } }
        ]
        tree.expandedIds = ['src']
      </script>
    </PondDemo>
  </Fragment>

  <Fragment slot="code">

```html
<pond-table-tree caption="Project"></pond-table-tree>
<script type="module">
  const el = document.querySelector('pond-table-tree')
  el.columns = [
    { key: 'name', label: 'Name' },
    { key: 'type', label: 'Type' }
  ]
  el.nodes = [
    {
      id: 'src',
      cells: { name: 'src', type: 'Folder' },
      children: [{ id: 'a', cells: { name: 'app.ts', type: 'File' } }]
    }
  ]
</script>
```
